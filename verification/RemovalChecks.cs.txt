        public System.Collections.IEnumerator AddPiezaAsync<T>(string id, Il2CppSystem.Collections.Generic.IReadOnlyList<SlotDeMaterialDeRopa> materials,
            Il2CppSystem.Action<T> done, bool hidden)
        {
            Pending.Enqueue(() =>
            {
                if (FailNext) { FailNext = false; done(default); return; }
                var slots = new Il2CppSystem.Collections.Generic.List<SlotDeMaterialDeRopa>();
                foreach (var slot in materials) slots.Add(slot.Clone());
                if (slots.Count == 0) slots.Add(ClothingMaterials.Create("default", 0, new Color()));
                var piece = new PiezaDeRopaBase { name = id, materialesData = slots, skinnedMeshRenderer = new() };
                piezasPuestasPorId[id] = piece;
                done((T)(object)piece);
            });
            yield break;
        }
        public void CompleteNext() { if (Pending.Count != 0) Pending.Dequeue()(); }
    }
